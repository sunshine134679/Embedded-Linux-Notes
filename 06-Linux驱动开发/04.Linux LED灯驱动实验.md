# Linux LED灯驱动实验

## 一、内存映射

1. 逻辑LED灯实验就是操作6ULL的寄存器。

   

2. Linux驱动开发也可以操作寄存器，但Linux下不能直接对寄存器的物理地址进行读写操作，比如寄存器A物理地址为 0X10101010。在裸机的时候可以直接对 0X10101010 这个物理地址进行操作，但是Linux不行，因为Linux会使能`MMU`。

   

3. 在Linux里面操作的都是虚拟地址，所以需要先得到0X10101010这个物理地址对应的虚拟地址。而对于物理地址和虚拟地址之间的转换，有两个函数可以实现：`ioremap` 和 `iounmap`。

   - 获得物理地址对应的虚拟地址使用 `ioremap`

     ```c
     ioremap(cookie,size);
     ```

     其中，第一个参数就是物理地址起始大小，第二个参数就是要转化的**字节**数量。

     例如，对物理地址0x10101010的开始10个地址进行转换：

     ```c
     va = ioremap(0X10101010,10);
     ```

     其中 `va` 就是 “Virtual address“（虚拟地址）。

     

   - 卸载驱动时释放虚拟地址使用`iounmap`

     ```c
     iounmap(va);
     ```

## 三、驱动程序编写

1. 初始化LED的时钟、IO、CPIO等等。

2. 初四还完成以后进行测试，但是如果你烧写或者用的是正点原子提供的Linux内核，这个时候LED灯默认被设置为了心跳灯，必须关闭心跳灯。

3. Linux下的读、写函数：

   ```c
   read(c);		//c即要读的地址（虚拟地址，不是物理地址！！！！）
   writel(v,c);	//v代表要写入的数据，c仍旧是要读的地址（虚拟地址）
   ```

   

4. 一下是驱动源码：

   ```c
   #include <linux/module.h>
   #include <linux/kernel.h>
   #include <linux/init.h>
   #include <linux/proc_fs.h>
   #include <linux/uaccess.h>
   #include <linux/io.h>
   
   #define LED_MAJOR    200   //主设备号
   #define LED_NAME    "led"  //设备名
   
   //函数前置声明
   static int __init led_init(void);
   static void __exit led_exit(void);
   static int led_open(struct inode *inode, struct file *file);
   static ssize_t led_write(struct file *file, const char __user *buffer,
   				    size_t count, loff_t *pos);
   static int led_release(struct inode *inode, struct file *file);
   static void led_switch(unsigned char state);
   
   
   //字符设备操作集
   static struct file_operations led_fops = {
       .owner      = THIS_MODULE,
       .write      = led_write,
       .open       = led_open,
       .release    = led_release,
   };
   
   // 寄存器的物理地址
   #define CCM_CCGR1_BASE          (0X020c406c)
   #define SW_MUX_GPIO1_IO03_BASE  (0X020E0068)
   #define SW_PAD_GPIO1_IO03_BASE  (0X020E02F4)
   #define GPIO1_DR_BASE           (0X0209C000)
   #define GPIO1_GDIR_BASE         (0X0209C004)
   
   //地址映射后的虚拟地址指针
   static void __iomem *CCM_CCGR1;
   static void __iomem *SW_MUX_GPIO1_IO03;
   static void __iomem *SW_PAD_GPIO1_IO03;
   static void __iomem *GPIO1_DR;
   static void __iomem *GPIO1_GDIR;
   
   #define LEDOFF  0
   #define LEDON   1
   
   
   //入口
   static int __init led_init(void) {
       int ret = 0;
       unsigned int val = 0;
   
       //LED地址映射
       CCM_CCGR1 = ioremap(CCM_CCGR1_BASE,4);
       SW_MUX_GPIO1_IO03 = ioremap(SW_MUX_GPIO1_IO03_BASE,4);
       SW_PAD_GPIO1_IO03 = ioremap(SW_PAD_GPIO1_IO03_BASE,4);
       GPIO1_DR = ioremap(GPIO1_DR_BASE,4);
       GPIO1_GDIR = ioremap(GPIO1_GDIR_BASE,4);
   
   
       //初始化LED->使能GPIO1时钟
       val = readl(CCM_CCGR1);
       val &= ~(3 << 26);      //先清除以前配置的bit26、27
       val |= (3 << 26);       //bit26、27置一
       writel(val,CCM_CCGR1);
   
       //初始化LED->设置GPIO1_IO03复用和配置电气属性
       writel(0x5,SW_MUX_GPIO1_IO03);
       writel(0x10b0,SW_PAD_GPIO1_IO03);
   
       //初始化LED->将GPIO1_IO03设置为输出
       val = readl(GPIO1_GDIR);
       val &= ~(1 << 3);       //此处和上述使能时钟置一的话，实际上不需要清零，但是养成习惯最好
       val |= (1 << 3);
       writel(val,GPIO1_GDIR);
   
       //初始化LED->将输出默认值设为高电平（默认LED关闭）
       val = readl(GPIO1_DR);
       val &= ~(1 << 3);       //清零bit3，默认高电平，LED灯关闭
       writel(val,GPIO1_DR);
       
   
       //注册字符设备驱动
       ret = register_chrdev(LED_MAJOR,LED_NAME,&led_fops);    
       if(ret < 0) {
           printk("register chrdev failed!\r\n");
           return -EIO;
       }
   
       printk("led init\r\n");
       return 0;
   }
   
   //出口
   static void __exit led_exit(void) {
       unsigned int val = 0;
   
       //关灯
       val = readl(GPIO1_DR);
       val |= (1 << 3);       
       writel(val,GPIO1_DR);
   
       //注销地址映射
       iounmap(CCM_CCGR1);
       iounmap(SW_MUX_GPIO1_IO03);
       iounmap(SW_PAD_GPIO1_IO03);
       iounmap(GPIO1_DR);
       iounmap(GPIO1_GDIR);
   
       unregister_chrdev(LED_MAJOR,LED_NAME);      //注销字符设备驱动
   
       printk("led exit\r\n");
   }
   
   //注册驱动加载和卸载
   module_init(led_init);
   module_exit(led_exit);
   
   //版权信息
   MODULE_LICENSE("GPL");
   MODULE_AUTHOR("wangshaolong");
   
   
   //以下是字符操作集中所需的函数:
   static int led_open(struct inode *inode,struct file *file) {
       return 0;
   }
   
   static ssize_t led_write(struct file *file, const char __user *buffer,
   				    size_t count, loff_t *pos)
   {
       int ret = 0;
       unsigned char databuf[1];
   
       ret = copy_from_user(databuf,buffer,count);
   
       if(ret < 0) {
           printk("kernel write failed!\r\n");
           return -EFAULT;
       }
   
       //判断是开灯还是关灯
       if(databuf[0] == LEDON) {
           led_switch(LEDON);
       }
       else if(databuf[0] == LEDOFF) {
           led_switch(LEDOFF);
       }
   
       return 0;
   }
   
   static int led_release(struct inode *inode,struct file *file) {
       return 0;
   }
   
   //LED灯开关函数
   static void led_switch(unsigned char state) {
       unsigned int val = 0;
       if(state == LEDON) {
           val = readl(GPIO1_DR);
           val &= ~(1 << 3);       
           writel(val,GPIO1_DR);
       }
       else if(state == LEDOFF) {
           val = readl(GPIO1_DR);
           val |= (1 << 3);       
           writel(val,GPIO1_DR);
       }
   }
   ```

   



## 四、应用程序编写

以下是应用程序源码：

```c
#include <fcntl.h>
#include <stdio.h>
#include <unistd.h>
#include <stdlib.h>
#include <string.h>

//argc：应用程序参数个数
//argv[]:具体的参数内容，字符串形式
//使用方式 ： ./ledAPP.c（应用程序名称）  <filename> <0:1> , 0代表关灯，1代表开灯........
//例如：./ledAPP /dev/led 0     关灯
//     ./ledAPP /dev/led 1     开灯
int main(int argc,char *argv[]) {
    if(argc != 3) {
        printf("Error Usage!\r\n");
        return -1;
    }

    int fd;         //文件描述符
    int ret = 0;
    unsigned char databuf[1];
    char *filename;

    filename = argv[1];

    fd = open(filename,O_RDWR);
    if(fd < 0) {
        printf("open file %s failed!\r\n", filename);
        return -1;
    }
    
    databuf[0] = atoi(argv[2]);     //将字符转换为数字

    //write函数如果写入成功，返回实际写入的字节数（一定大于0）；写入失败，则返回错误（负值）
    //fd:文件描述符，指定要往哪个文件里面写数据
    //databuf：数据缓冲区，储存要写入的数据
    //sizeof(databuf):表示要写入的数据长度，这里就使用databuf的长度
    ret = write(fd,databuf,sizeof(databuf));
    if(ret < 0) {
        printf("LED Control failed!\r\n");
        close(fd);
        return -1;
    }

    close(fd);

    return 0;
}
```
